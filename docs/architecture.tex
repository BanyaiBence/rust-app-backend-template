% Architecture documentation template.
% Build with `make architecture` (docs/build.sh). The script renders docs/architecture/*.puml
% to build/figures/*.pdf and writes build/diagrams.tex, one section per diagram; this file only
% holds the title page, the overview and the layout macros.
%
% To adapt it: set \projectname and \projectsubtitle below and write the overview. Add a
% diagram by dropping a .puml file into docs/architecture/ (the file name sets the order);
% put optional prose for it in docs/architecture/<name>.tex.

% Page dimensions in portrait orientation, saved before any landscape page swaps them.
\newlength{\portraitwidth}
\newlength{\portraitheight}
\AtBeginDocument{%
  \setlength{\portraitwidth}{\textwidth}%
  \setlength{\portraitheight}{\textheight}%
}

% Room left for a section heading, a short paragraph and the caption.
\newcommand{\textallowance}{9\baselineskip}

% \diagram{file}{caption}: a figure scaled to fill the rest of a portrait page.
% Not a float, so it stays on the same page as its section text.
\newcommand{\diagram}[2]{%
  \begin{center}
    \includegraphics[width=\portraitwidth,
                     height=\dimexpr\portraitheight-\textallowance\relax,
                     keepaspectratio]{#1}
    \captionof{figure}{#2}
  \end{center}%
}

% \widediagram{file}{caption}: the same for use inside a landscape environment.
\newcommand{\widediagram}[2]{%
  \begin{center}
    \includegraphics[width=\portraitheight,
                     height=\dimexpr\portraitwidth-\textallowance\relax,
                     keepaspectratio]{#1}
    \captionof{figure}{#2}
  \end{center}%
}

% \architecturediagram{file}{title}{prose file or empty}: one section of build/diagrams.tex,
% on its own page. Diagrams clearly wider than tall get a landscape page.
\newsavebox{\diagrambox}
\newcommand{\architecturediagram}[3]{%
  \clearpage
  \sbox{\diagrambox}{\includegraphics{#1}}%
  \ifdim\wd\diagrambox>1.2\ht\diagrambox
    \begin{landscape}
      \section{#2}
      \if\relax\detokenize{#3}\relax\else\input{#3}\fi
      \widediagram{#1}{#2}
    \end{landscape}
  \else
    \section{#2}
    \if\relax\detokenize{#3}\relax\else\input{#3}\fi
    \diagram{#1}{#2}
  \fi
}
